export const midgardr_db_seed = Staff_rdb_schema('midgardr_db_seed', ({ self, field }: any) => ({
    name: 'midgardr_db',
    dbType: 'sqlite',
    memory: true,
    path: './db.db',
    sdk: {
        id: field.uuid().primaryKey(),
        name: field.string().unique(),
        description: field.string().nullable(),
    },
    library: {
        id: field.uuid().primaryKey(),
        sdkId: field.uuid(),
        name: field.string().unique(),
        description: field.string().nullable(),
    }.oneToMany('sdkId', 'sdk', 'id'),
    category: {
        id: field.uuid().primaryKey(),
        libraryId: field.uuid(),
        name: field.string(),
        description: field.string().nullable(),
    }.oneToMany('libraryId', 'library', 'id'),
    component: {
        id: field.uuid().primaryKey(),
        categoryId: field.uuid(),
        name: field.string(),
        value: field.string().unique(),
        importPath: field.string(),
        multiImport: field.string().nullable(),
        path: field.string(),
        premium: field.boolean().default(false),
        source: field.string().nullable(),
        tags: field.json().nullable(),
        features: field.json().nullable(),
        dependencies: field.json().nullable(),
        basicUsage: field.string().nullable(),
        usage: field.string().nullable(),
        props: field.json().nullable(),
        description: field.string().nullable(),
        status: field.string().nullable(),
        lastUpdated: field.timestamp().nullable(),
    }.oneToMany('categoryId', 'category', 'id'),
    doc:  {
        id: field.uuid().primaryKey(),
        categoryId: field.uuid(),
        componentId: field.uuid().nullable(),
        title: field.string(),
        slug: field.string().unique(),
        content: field.string(),
        tags: field.json().nullable(),
        createdAt: field.timestamp().default('now'),
        updatedAt: field.timestamp().default('now'),
    }.oneToMany('categoryId', 'category', 'id'),
}))

        qsCustom_mobile: useState(false),
        isMobile: useState(false),

        leftWidthValue: useState(DEFAULT_WIDTH),
        rightWidthValue: useState(DEFAULT_WIDTH_RIGHT),

        bottomHeightValue: useState(DEFAULT_HEIGHT_BOTTOM),
        customSidebarsWidthValue: useState(DEFAULT_HEIGHT_CUSTOM),
        customSidebarsHeightValue: useState(DEFAULT_WIDTH_CUSTOM),

        qs_variant: DEFAULT_VARIANT,
        qsRight_variant: DEFAULT_VARIANT_RIGHT,
        qsTop_variant: DEFAULT_VARIANT_TOP,
        qsBottom_variant: DEFAULT_VARIANT_BOTTOM,
        qsCustom_variant: DEFAULT_VARIANT_CUSTOM,
    },
    toggles: {
        qs: {
            onClick: useCallback(() => {
                if (self.states.isMobile) {
                    self.states.qs_mobile((p: any) => !p);
                } else {
                    self.states.qs((p: any) => {
                        const next = !p;
                        document.cookie = `${QS_COOKIE_NAME}=${next}; path=/; max-age=${QS_COOKIE_MAX_AGE}`;
                        return next;
                    });
                }
            }, [self.states.isMobile])
        },
        qsRight: {
            onClick: useCallback(() => {
                if (self.states.isMobile) {
                    self.states.qsRight_mobile((p: any) => !p);
                } else {
                    self.states.qsRight((p: any) => {
                        const next = !p;
                        document.cookie = `${QS_COOKIE_NAME_RIGHT}=${next}; path=/; max-age=${QS_COOKIE_MAX_AGE}`;
                        return next;
                    });
                }
            }, [self.states.isMobile])
        }
    },
    layout_toggles: {
        qs: '',
    },
    persistence: {

    },
    shortcuts: {
        qs: useEffect(() => {

            window.addEventListener('keydown', handler);
            return () => window.removeEventListener('keydown', handler);
        }, [self.toggles.qs]),
        qsRight: useEffect(() => {
            const handler = (e: KeyboardEvent) => { if (e.key === QS_KEYBOARD_SHORTCUT_RIGHT && (e.metaKey || e.ctrlKey)) { e.preventDefault(); self.toggles.qsRight(); } };
            window.addEventListener('keydown', handler);
            return () => window.removeEventListener('keydown', handler);
        }, [self.toggles.qsRight])

            } else if (side === 'right') {

Ward_lvl2('auth', ({ self, staff, routes, sentinel, kvdb, redirect }: any) => ({
    create_memory_storage: () => {
        const persistent = new Map<string, unknown>()
        const memory = new Map<string, unknown>()
        const watchers = new Map<string, Set<(v: unknown) => void>>()
        const ttls = new Map<string, ReturnType<typeof setTimeout>>()

        function _notify(key: string, value: unknown): void { watchers.get(key)?.forEach(fn => fn(value)) }

        function _scheduleExpiry(key: string, ms: number, store: Map<string, unknown>): void {
            const existing = ttls.get(key)
            if (existing) clearTimeout(existing)
            const t = setTimeout(() => { store.delete(key); ttls.delete(key); }, ms)
            ttls.set(key, t)
        }

        return {
            async get(key: string,) { return persistent.get(key) ?? null },
            async set(key: string, value: any) { persistent.set(key, value); _notify(key, value); },
            async delete_entry(key: string,) {
                persistent.delete(key);
                const t = ttls.get(key);
                if (t) { clearTimeout(t); ttls.delete(key) };
                _notify(key, null);

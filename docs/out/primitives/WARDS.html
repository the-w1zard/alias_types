    create_memory_storage: () => {
        const persistent = new Map<string, unknown>()
        const memory = new Map<string, unknown>()
        const watchers = new Map<string, Set<(v: unknown) => void>>()
        const ttls = new Map<string, ReturnType<typeof setTimeout>>()

        function _notify(key: string, value: unknown): void { watchers.get(key)?.forEach(fn => fn(value)) }

        function _scheduleExpiry(key: string, ms: number, store: Map<string, unknown>): void {
            const existing = ttls.get(key)
            if (existing) clearTimeout(existing)
            const t = setTimeout(() => { store.delete(key); ttls.delete(key); }, ms)
            ttls.set(key, t)
        }

        return {
            async get(key: string,) { return persistent.get(key) ?? null },
            async set(key: string, value: any) { persistent.set(key, value); _notify(key, value); },
            async delete_entry(key: string,) {
                persistent.delete(key);
                const t = ttls.get(key);
                if (t) { clearTimeout(t); ttls.delete(key) };
                _notify(key, null);
            },
            async getMemory(key: string,) { return memory.get(key) ?? null; },
            async setMemory(key: string, value: any) { memory.set(key, value); },
            async deleteMemory(key: string,) { memory.delete(key); },
            // apply to whichever store has the key
            async expire(key: string, ms: any) { const store = persistent.has(key) ? persistent : memory; _scheduleExpiry(key, ms, store); },
            async watch(key: string, handler: any) {
                if (!watchers.has(key)) watchers.set(key, new Set())
                watchers.get(key)!.add(handler)
                return persistent.get(key) ?? null
            },
            _dump() {
                const out: Record<string, unknown> = {};
                persistent.forEach((v, k) => { out[k] = v });
                return out;
            },
        }

export const AdminAuthGolem = Golem('admin_auth_golem', ({ self, tw, staff, sentinel }: any) => ({
    // 1. The Policy: Does this pass inspection?
    enforce: async (reqContext: any) => {
        const session = await sentinel.session_mgmt.verify(reqContext.cookies);
        return session && session.role === 'ADMIN';
    },

    // 2. The Breach: What happens when enforcement fails?
    breach: (reqContext: any) => {
        // Automatically hook into your routing system to bounce the user
        return {
            action: 'REDIRECT',
            destination: "/auth/login",
            toast: "The Golem denies entry. Admin access required."
        };
    },

    // 3. The Enrichment: Mutate/inject data for the Ward to use
    enrich: async (reqContext: any) => {
        const user = await staff.rdb.table.users.findById(reqContext.userId);
        // Bind the validated user to the global scope for this specific request cycle
        return { activeAdmin: user };
    }
}));
export const AdminSettingsRoute = APIWard("/api/admin/settings", ({ self, admin_auth_golem, rate_limit_golem }: any) => ({
    // Summon Golems to guard this route. They execute in order.
    // If rate_limit_golem fails, it never bothers hitting the DB for admin_auth_golem.
    golems: [
        rate_limit_golem,
        admin_auth_golem
    ],

    return: (req: any) => {
        // By the time the code reaches here, the Golems have guaranteed safety.
        // Furthermore, the auth Golem enriched the context with \`activeAdmin\`!
        const admin = req.context.activeAdmin;
        const v = {
            status: 200,
            data: `Welcome to the sanctum, ${admin.name}`
        }
        return v;
    }
}));

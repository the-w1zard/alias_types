export class Chain<T = A> {
    public initial_value: T
    public id: A
    constructor(value?: A, id?: A) { this.initial_value = value; this.id = id ? id : _sid() }

    private _use() {
        return root.DISRUPTOR.useCell(`re_chain_${this.id}`, this.initial_value,)
    }

    /* ---------- branch logic ---------- */
    when(p: boolean | ((v: T) => boolean)) {
        const [v] = this._use()
        const ok = typeof p === 'function' ? (p as (v: T) => boolean)(v) : p
        return {
            then: (fn: (v: T) => T | Promise<T>) => (ok ? fn(v) : undefined),
            otherwise: (fn: (v: T) => T | Promise<T>) => (ok ? v : fn(v)),
        }
    }
    if(p: boolean | ((v: T) => boolean)) { return this.when(p) }

    switch() {
        const self = this
        const [v0, setV] = this._use()
        let matched = false
        let current: A = v0
        const view: {
            case: <V = A>(m: V, fn: (v: T) => T | Promise<T>) => A
            default: (fn: (v: T) => T | Promise<T>) => Chain<T>
            done: () => T
        } = {
            case: (m, fn) => {
                if (!matched && current === m) {
                    matched = true
                    current = fn(current) as T
                    setV(current)
                    self.initial_value = current
                }
                return view
            },
            default: (fn) => {
                if (!matched) {
                    current = fn(current) as T
                    setV(current)
                    self.initial_value = current
                }
                return self
            },
            done: () => current,
        }
        return view
    }

    guard(p: (v: T) => boolean, e: A = new Error('guard failed')) {
        const [v] = this._use()
        if (!p(v)) throw e
        return this
    }

    /* ---------- mutation steps ---------- */
    set(v: T | ((c: T) => T)) {
        const [current, setV] = this._use()
        const next = (typeof v === 'function' ? (v as A)(current) : v) as T
        setV(next)
        this.initial_value = next
        return this
    }
    patch(p: Partial<T>) {
        const [current, setV] = this._use()
        const next = { ...(current as A), ...(p as A) } as T
        setV(next)
        this.initial_value = next
        return this
    }
    merge(o: A) {
        const [current, setV] = this._use()
        const next = { ...(current as A), ...(o as A) } as T
        setV(next)
        this.initial_value = next
        return this
    }
    tap(fn: (v: T) => A) {
        const [v] = this._use()
        fn(v)
        return this
    }
    get<K extends keyof T>(k: K): Chain<T[K]> {
        const [v] = this._use()
        return new Chain(v?.[k] as T[K])
    }

    /* ---------- async ---------- */
    then(fn: (v: T) => T | Promise<T>) {
        const [v] = this._use()
        return Promise.resolve(v).then(fn).then((next: A) => new Chain(next) as A)
    }
    catch(fn: (e: A) => T) {
        const [v] = this._use()
        return Promise.resolve(v).catch(fn).then((next: A) => new Chain(next))
    }
    retry(fn: (v: T) => Promise<T>, opts: { attempts?: number; baseDelayMs?: number } = {}) {
        const attempts = opts.attempts ?? 3, base = opts.baseDelayMs ?? 2000
        const [v0] = this._use()
        return (async () => {
            let lastErr: A
            for (let i = 0; i < attempts; i++) {
                try { return new Chain(await fn(v0)) }
                catch (e) { lastErr = e; if (i < attempts - 1) await new Promise(r => setTimeout(r, base * 2 ** i)) }
            }
            throw lastErr
        })()
    }
    delay(ms: number) { return new Promise<Chain<T>>(r => setTimeout(() => r(this), ms)) }
    throw(e: A): never { throw e }

    done(): T {
        const [v] = this._use()
        return (typeof v === 'function' ? (v as Function)() : v) as T
    }
}

Core Concept

The Chain<T> class acts as a fluent builder / monad pattern for state manipulation. Instead of writing imperative updates or nested callbacks, it allows you to chain synchronous transformations, asynchronous steps, conditional branches, and error handling sequentially against a reactive cell (root.DISRUPTOR.useCell).

Key Architectural Groups

1. Initialization & Scoping

  • constructor(value?: A, id?: A): Initializes the chain with an optional starting value and ID (auto-generated via _sid() if omitted).
  • _use(): Internal helper that hooks into the underlying distributed reactive system to fetch the latest state tuple ([value, setter]) for re_chain_${this.id}.
  • get<K>(k: K): Narrows the current chain context down to a specific nested property, returning a new Chain<T[K]>.

2. Conditional Branching & Guarding

  • **when(p) / if(p)**: Evaluates a boolean or a predicate function against the current state value. Returns a builder exposing .then(fn) and .otherwise(fn) for inline branching.
  • switch(): Provides a declarative pattern-matching syntax. Allows chaining .case(m, fn) blocks, a fallback .default(fn), and a .done() terminator to compute and commit state transitions.
  • guard(p, e): Acts as a defensive assertion. If the predicate p(v) fails, it throws an error (e, defaulting to 'guard failed'); otherwise, it passes this through for continued chaining.

3. State Mutations & Side Effects

  • set(v): Overwrites the current state or applies a functional updater (c: T) => T, committing it back to the underlying cell.
  • **patch(p) / merge(o)**: Shallow-merges partial object updates into the current state.
  • tap(fn): Executes a side-effect (e.g., logging or triggering an external event) using the current value without altering the state itself.

4. Asynchronous Flow & Resilience

  • then(fn): Handles asynchronous transformations by returning a Promise that resolves into a new Chain instance.
  • catch(fn): Gracefully catches errors within an asynchronous chain and transforms them.
  • retry(fn, opts): Implements built-in resilience for async operations, running up to a configurable number of attempts with exponential backoff (baseDelayMs * 2 ** i).
  • delay(ms): Pauses the chain execution for a specified duration using a setTimeout promise before returning this.
  • done(): Terminates the chain and extracts the final underlying value.

export class ReChain<T = A> {
    public initial_value: A
    public id: A
    constructor(value?: A, id?: A) { this.initial_value = value; this.id = id ? id : _sid() }

    private _use() {
        return root.DISRUPTOR.useCell(this.initial_value, `re_chain_${this.id}`)
    }
    // the ONLY place a value changes hands — a fresh chain, nothing mutated.
    // write=true means this transformation broadcasts to the store + bus.
    private step(next: A, write = false): ReChain<A> {
        if (write) {
            const [, setV] = this._use();
            setV(next);
        }
        return next instanceof ReChain ? next : new ReChain(next, this.id);
    }
    // when/if/switch return a branch view: then and otherwise can BOTH be
    // written (only one fires, like Chain), and each call returns a fresh
    // ReChain — nothing mutated, order-independent. ✔ ✘
    private branch(passed: boolean) {
        const step = (next: A, write = false) => this.step(next, write);
        return {
            then: (fn: A) => (passed ? step(fn(this.done()), true) : this),
            otherwise: (fn: A) => (passed ? this : step(fn(this.done()), true)),
            else: (fn: A) => (passed ? this : step(fn(this.done()), true)),
            done: () => this.done(),
        };
    }
    public when(pred: (v: T) => boolean) {
        const [v] = this._use()
        log(`[RE_CHAIN][WHEN]`, { success: '✔', VALUE: v, PRED: pred, }, t.info)
        return this.branch(!!pred(v));
    }
    public if(pred: (v: T) => boolean) {
        const [v] = this._use()
        log(`[RE_CHAIN][IF]`, { success: '✔', VALUE: v, PRED: pred, }, t.info)
        return this.branch(!!pred(v));
    }
    public switch() {
        const [v, setV] = this._use()
        let matched = false;
        let out: ReChain<A> = this;
        log(`[RE_CHAIN][SWITCH]`, { success: '✔', VALUE: v, }, t.info)
        const self = this
        return {
            case(val: A, fn: A) {
                if (!matched && (v === val || Object.is(v, val))) {
                    matched = true;
                    const next = fn(v);
                    setV(next); // write-through ✔
                    out = new ReChain(next, self.id);
                }
                return self;
            },
            default(fn: A) {
                if (!matched) {
                    const next = fn(v);
                    setV(next); // write-through ✔
                    out = new ReChain(next, self.id);
                }
                return self;
            },
            done: () => out.done(),
        };
    }
    public gate(pred: (v: T) => boolean): this {
        const [v] = this._use()
        if (!pred(v)) {
            const err: A = new Error('[Chain.gate] predicate failed');
            err.value = v;
            log(`[RE_CHAIN][GATE]`, { success: '✘', VALUE: v, PRED: pred, ERROR: err }, t.error)
            throw err;
        }
        log(`[RE_CHAIN][GATE]`, { success: '✔', VALUE: v, PRED: pred, }, t.info)
        return this;
    }
    public guard(pred: (v: T) => boolean, onFail?: (v: T) => A) {
        const [v, setV] = this._use()
        if (pred(v)) return this;
        const fallback = onFail ? onFail(v) : v;
        setV(fallback); // write-through ✔
        return this.step(fallback);
    }
    public tap(fn: (v: T) => A) {
        const [v] = this._use()
        fn(v);
        return this;
    }
    public set(k: string, val: A) {
        const [v, setV] = this._use()
        const next = { ...v, [k]: val };
        setV(next); // write-through ✔
        return this.step(next);
    }
    public patch(obj: object) {
        const [v, setV] = this._use()
        const next = { ...v, ...obj };
        setV(next); // write-through ✔
        return this.step(next);
    }
    public get(k: string) {
        const [v] = this._use()
        return this.step(v?.[k]);
    }
    public merge(other: A) {
        const [v, setV] = this._use()
        const next = { ...v, ...(other?.initial_value ?? other) };
        setV(next); // write-through ✔
        return this.step(next);
    }
    public delay(ms: number): Promise<ReChain<T>> {
        return new Promise((resolve) => setTimeout(() => resolve(this), ms));
    }
    public throw(err: A): never {
        throw err instanceof Error ? err : new Error(String(err));
    }
    public retry(attempts: number, delayMs = 100): Promise<T> {
        return new Promise<T>((resolve, reject) => {
            let n = 0;
            const attempt = () => {
                const [v] = this._use()
                const out = typeof v === 'function' ? (v as Function)() : v;

                if (out !== undefined && out !== null) {
                    Promise.resolve(out).then(resolve, reject);
                    return;
                }

                if (n++ >= attempts) {
                    reject(new Error('[Chain.retry] exhausted'));
                    return;
                }

                setTimeout(attempt, delayMs);
            };
            attempt();
        });
    }
    public catch<R = never>(onRejected: (e: A) => R | PromiseLike<R>): Promise<T | R> {
        return this.then(null, onRejected);
    }
    public finally(fn: () => A): this {
        fn();
        return this;
    }
    public then<R = T>(onFulfilled?: ((v: T) => R | PromiseLike<R>) | null, onRejected?: ((e: A) => R | PromiseLike<R>) | null,): Promise<T | R> {
        const [v] = this._use()
        const raw = typeof v === 'function' ? (v as Function)() : v;
        return Promise.resolve(raw).then(onFulfilled as A, onRejected);
    }
    public done(): T {
        const [v] = this._use()
        return (typeof v === 'function' ? (v as Function)() : v) as T;
    }
}

Core Enhancements & Architecture

Compared to the initial version, ReChain introduces three major structural upgrades:

  • Centralized State Progression (step): A single private step(next, write) method handles value propagation. It ensures that every transformation cleanly returns a fresh ReChain instance tied to the same underlying ID, keeping operations predictable and side-effect contained.
  • Built-in Observability (log): Integrated telemetry tags ([RE_CHAIN][WHEN], [RE_CHAIN][SWITCH], [RE_CHAIN][GATE], etc.) track success/failure states, predicates, and values for every major decision point.
  • Flexible Branching & Fallbacks: Expanded branch controls (then, otherwise, else) and a forgiving guard pattern that supports custom fallback values with write-through persistence.

Key Functional Groups

1. Core Mechanics & Flow Control

  • constructor(value?, id?): Initializes the chain with a starting value and an ID (auto-generated via _sid() if missing).
  • _use(): Connects to the distributed reactive store via root.DISRUPTOR.useCell(...) to fetch the current value and setter.
  • step(next, write): The gatekeeper method. If write is true, it broadcasts the update to the underlying store; it then wraps the result in a fresh ReChain instance.

2. Conditional Branching & Pattern Matching

  • **when(pred) / if(pred)**: Evaluates a predicate against the current state value, logs the result, and returns a unified branch object.
  • branch(passed): Powers inline conditional logic, exposing .then(fn), .otherwise(fn), and .else(fn) to execute mutually exclusive paths while preserving chain fluidity.
  • switch(): Provides a declarative pattern-matching utility. It evaluates .case(val, fn) blocks and a .default(fn) fallback, applying write-through updates upon a match.

3. Validation & Safety (gate vs guard)

  • gate(pred): A strict assertion. If the predicate fails, it logs an error containing the offending value and throws an explicit Error with value context attached.
  • guard(pred, onFail): A resilient alternative. If the predicate fails, it executes an optional onFail fallback, writes the fallback value through to the store, and continues the chain.

4. Mutations & Property Extraction

  • set(k, val): Updates a specific key in an object state, writing the change through and returning a stepped chain.
  • **patch(obj) / merge(other)**: Shallow-merges partial objects or other chain initial values into the current state.
  • get(k): Extracts a nested property by key and steps into a new chain context.
  • tap(fn): Executes an external side-effect with the current value without altering the state.

5. Asynchronous & Promise-Like Lifecycle

  • then(onFulfilled, onRejected): Bridges synchronous chain values directly into standard JavaScript Promise resolution flows.
  • catch(onRejected): Handles promise rejections or errors cleanly within the chain sequence.
  • retry(attempts, delayMs): Automatically retries resolving null or undefined/failing state outputs up to a specified attempt count with a fixed delay.
  • delay(ms): Pauses chain execution for a specified duration using a timeout promise.
  • finally(fn): Executes a cleanup or final callback function and passes the chain through.
  • done(): Terminates the chain pipeline and returns the raw final value.

While both classes use a fluent builder pattern, ReChain leans heavily into functional immutability, whereas the original Chain relies on direct, in-place mutation.

Here is how that plays out in ReChain:

  • Fresh Instances on Every Step: Every operation that changes or navigates state (step, get, set, patch, merge, guard) wraps the result in a brand new ReChain instance (new ReChain(...)).
  • Controlled Side-Effects: The original Chain mutates this.initial_value in place on almost every method call. ReChain isolates all writes strictly through step(next, write), ensuring that intermediate references remain clean, predictable, and order-independent.
  • Safer Branching: In ReChain, branching methods (when, if, switch) evaluate against the current state snapshot and return distinct branches without polluting the root chain instance unless an explicit write-through occurs.

It’s a much safer, more composable design for a distributed reactive environment where multiple processes or UI bindings might be observing or branching off the same state tree!
